<script>
function filterUniversities() {
    let search = document.getElementById("searchBar").value.toLowerCase();
    let lang = document.getElementById("languageFilter").value;
    let type = document.getElementById("typeFilter").value;
    let cards = document.querySelectorAll(".card");

    cards.forEach(card => {
        let name = card.querySelector("h2").innerText.toLowerCase();
        let cardLang = card.getAttribute("data-lang");
        let cardType = card.getAttribute("data-type");

        let matchSearch = name.includes(search);
        let matchLang = (lang === "all" || cardLang.includes(lang));
        let matchType = (type === "all" || cardType === type);

        card.style.display = (matchSearch && matchLang && matchType) ? "block" : "none";
    });
}
</script>
